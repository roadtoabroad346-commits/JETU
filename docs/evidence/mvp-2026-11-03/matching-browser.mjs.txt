import { createServer } from 'vite';
import { mkdir, writeFile } from 'node:fs/promises';
import { chromium } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

 }

   if(route.request().method()==='GET') {initialLoads++; await route.fulfill(initialLoads===1 ? {status:503,json:{error:{code:'INTERNAL_ERROR'}}}:{json:{data}});return;}
   await route.fulfill({status:401,json:{error:{code:'UNAUTHENTICATED'}}});
 });

 await page.getByRole('button',{name:'Try again',exact:true}).click();

 await page.getByRole('button',{name:'Run matching',exact:true}).click();

 if(errors.length)throw Error(errors.join('\n'));
 console.log('PASS: actual MatchingWorkspace with mocked API: validation/focus, POST failure, retained draft, POST retry, success, initial GET recovery, session expiry, 320/768/1440 layout and axe. Not hosted Auth/REST evidence.');
} finally {await browser.close();await server.close();}
